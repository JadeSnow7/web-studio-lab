async function freePort(): Promise<number> {
  return new Promise((resolve, reject) => {
    const server = createServer();
    server.once('error', reject);
    server.listen(0, '127.0.0.1', () => {
      const address = server.address();
      server.close(() => resolve(typeof address === 'object' && address ? address.port : 0));
    });
  });
}

async function project(name: string, app?: string) {
  const directory = path.join(await mkdtemp(path.join(tmpdir(), `wsl-app-control-${name}-`)), 'project');
  await cp(template, directory, {
    recursive: true,
    filter: (source) => !/[\\/](node_modules|dist|\.data|\.test-data)([\\/]|$)/.test(source.slice(template.length)),
  });
  await symlink(path.join(template, 'node_modules'), path.join(directory, 'node_modules'), 'dir');
  if (app) await copyFile(app, path.join(directory, 'src/App.tsx'));
  return directory;
}

async function npm(directory: string, ...args: string[]) {
  return execute('npm', args, {
    cwd: directory,
    env: { ...process.env, WSL_AGENT_TOKEN: '', WSL_AGENT_URL: '' },
    timeout: 240_000,
    maxBuffer: 16 * 1024 * 1024,
  });
}

async function serve(directory: string, port: number, options: { token: string | null; appInstanceId: string; dataDir: string }) {
  const child = spawn(process.execPath, ['dist/server/index.js'], {
    cwd: directory,
    env: {
      ...process.env,
      HOST: '127.0.0.1',
      PORT: String(port),
      WSL_DATA_DIR: options.dataDir,
      WSL_DEV_SESSIONS: '1',
      WSL_WORKSPACE_ID: 'harness-space',
      WSL_ENVIRONMENT_ID: 'harness',
      WSL_PROJECT_ID: 'harness-project',
      WSL_APP_INSTANCE_ID: options.appInstanceId,
      WSL_AGENT_TOKEN: options.token ?? '',
    },
    stdio: ['ignore', 'pipe', 'pipe'],
  });
  let output = '';
  await new Promise<void>((resolve, reject) => {
    const timer = setTimeout(() => reject(new Error(`server not ready: ${output}`)), 30_000);
    child.stdout!.on('data', (chunk: Buffer) => {
      output += chunk.toString();
      if (output.includes('"app-ready"')) {
        clearTimeout(timer);
        resolve();
      }
    });
    child.stderr!.on('data', (chunk: Buffer) => (output += chunk.toString()));
    child.once('exit', (code) => reject(new Error(`server exited ${code}: ${output}`)));
  });
  return child;
}

async function stop(child: ChildProcess) {
  if (child.exitCode !== null) return;
  const exited = new Promise((resolve) => child.once('exit', resolve));
  child.kill('SIGTERM');
  await exited;
}

let bundle: string | null = null;
async function harnessBundle() {
  if (bundle) return bundle;
  const outDir = await mkdtemp(path.join(tmpdir(), 'wsl-app-control-harness-'));
  await build({
    configFile: false,
    root,
    logLevel: 'warn',
    build: {
      ssr: path.join(root, 'e2e/fixtures/app-control-harness-main.ts'),
      outDir,
      target: 'node22',
      minify: false,
      rollupOptions: { external: ['electron'], output: { format: 'cjs', entryFileNames: 'main.cjs' } },
    },
    ssr: { noExternal: true },
  });
  bundle = path.join(outDir, 'main.cjs');
  return bundle;
}

async function launch(url: string, token: string, appInstanceId: string, evidence: string) {
  const executablePath = createRequire(path.join(root, 'apps/desktop/package.json'))('electron') as unknown as string;
  const app = await _electron.launch({
    executablePath,
    args: [await harnessBundle(), `--user-data-dir=${await mkdtemp(path.join(tmpdir(), 'wsl-app-control-profile-'))}`],
    env: { ...process.env, WSL_HARNESS_CONFIG: JSON.stringify({ url, token, appInstanceId, evidence }) },
  });
  await expect
    .poll(() => app.evaluate(() => (globalThis as unknown as { __harness?: { ready(): boolean } }).__harness?.ready() ?? false), {
      timeout: 30_000,
    })
    .toBe(true);
  return app;
}

/** 有界关闭：超时后只结束本测试启动的 Electron 进程。 */
async function closeApp(app: ElectronApplication) {
  const child = app.process();
  await Promise.race([app.close().catch(() => undefined), new Promise((resolve) => setTimeout(resolve, 10_000))]);
  if (child.exitCode === null && child.signalCode === null) child.kill('SIGKILL');
}

const call = (app: ElectronApplication, tool: string, args: Record<string, unknown> = {}, callId?: string) =>
  app.evaluate((_app, input) => (globalThis as unknown as { __harness: Harness }).__harness.call(input.tool, input.args, input.callId), {
    tool,
    args,
    callId,
  });
const harness = <T>(app: ElectronApplication, method: string, ...args: unknown[]) =>
  app.evaluate(
    (_app, input) =>
      (globalThis as unknown as { __harness: Record<string, (...values: unknown[]) => unknown> }).__harness[input.method]!(...input.args),
    { method, args },
  ) as Promise<T>;
const code = (result: AppControlResult) => (result.ok ? 'ok' : result.error.code);
const data = <T = Record<string, unknown>>(result: AppControlResult): T => {
  if (!result.ok) throw new Error(JSON.stringify(result.error));
  return result.data as T;
};
async function record(name: string, value: unknown) {
  results[name] = value;
  await mkdir(evidenceRoot, { recursive: true });
  await writeFile(path.join(evidenceRoot, 'results.json'), JSON.stringify(results, null, 2) + '\n');
}
