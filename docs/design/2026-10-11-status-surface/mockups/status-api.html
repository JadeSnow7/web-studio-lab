export interface StatusFact {
  key: string;          // domain:entityId:facet，并发条件不重键
  domain: StatusDomain;
  facet: string;        // 'state' | 'load-error' | 'agent-control' | …
  scope: StatusScope;
  phase: StatusPhase;   // StatusDot 形状
  kind: 'progress' | 'control' | 'unresolved' | 'problem'
      | 'confirmation' | 'steady' | 'capability';
  priority: number;     // 只在同一通道内比较
  label: string;        // ≤120 字
  detail?: string;      // ≤300 字
  action?: StatusAction;
  instanceToken: string; // 实例更换 → 确认失效
  raw: StatusRaw;       // 按域白名单，≤2 KiB
}

projectFacts(input: FactInput): StatusFact[]          // ① Main + Renderer
presentStatus(facts, presentation): StatusView[]       // ② 仅 Renderer
useStatusView(scope) → { persistent, problem, more, notice, ambient }
